cmake_minimum_required(VERSION 3.20)
project(sqlite_sync VERSION 1.0.0 LANGUAGES CXX)

find_package(SQLite3 REQUIRED)
add_library(sync_core src/FileHasher.cpp src/DatabaseValidator.cpp src/Json.cpp src/SyncManager.cpp)
target_compile_features(sync_core PUBLIC cxx_std_20)
set_target_properties(sync_core PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(sync_core PUBLIC src)
target_link_libraries(sync_core PUBLIC SQLite::SQLite3)
if(MSVC)
  target_compile_options(sync_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sync_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(sqlite_sync src/main.cpp)
target_link_libraries(sqlite_sync PRIVATE sync_core)
if(MINGW)
  target_link_options(sqlite_sync PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(sync_tests tests/sync_tests.cpp)
  target_link_libraries(sync_tests PRIVATE sync_core)
  add_test(NAME sync_tests COMMAND sync_tests)
  add_test(NAME cli_help COMMAND sqlite_sync --help)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:sqlite_sync>)
  endif()
endif()
